# ── prune_v2 單調掃描（取代粗測+細搜；詳見 base_config.PruneV2Config / prune_v2_search.py）──
# t_max_ratio：搜尋上界 = 此比例 × 總神經元（1.0=整個模型）。過往最佳格都在 ≤20% T，
#              設 0.3~0.4 可把解析度集中在低 T、不浪費高 T。
# sweep_steps：把 [0, 上界] 切成幾個大步；step = (t_max_ratio×總神經元) / sweep_steps（★基於上界切）
# k_ratio_start：起始 K 占比（初始 K = round(k_ratio_start × 起始 T)）
# k_decay：翻轉夠高時 K ×= 此值（整數衰減、不會浮點消失）。易翻模型用接近 1（0.9）會跑很多格；0.5~0.7 較省。
# flip_low：翻轉率低於此值 → 放大 T（K 等比放大維持比例）；否則 → 衰減 K
t_max_ratio       = 1.0
sweep_steps       = 10
k_ratio_start     = 0.5
k_decay           = 0.9
flip_low          = 0.3
